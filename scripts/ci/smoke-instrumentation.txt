# JVM instrumentation on a real server (standard JVM, production profile).
# The server starts WITHOUT -javaagent; the agent is attached to the running JVM with the
# launcher in the agent jar (java -jar lemon-livepatch-agent.jar attach <pid>). Then a patch
# built in CI from changed Paper sources (org.spigotmc.TicksPerSecondCommand) is applied by
# retransformation through Lemon's transformer, read back, and rolled back.
fork instrumentation status
@sleep 2
fork instrumentation classes TicksPerSecond
@sleep 3
@shell java -jar "$AGENT_JAR" attach "$SERVER_PID"
@sleep 2
fork instrumentation status
@sleep 3
tps
fork instrumentation classes spigotmc.TicksPerSecond paper
@sleep 3
fork instrumentation classes org.spigotmc.*Command
@sleep 3
fork instrumentation inspect org.spigotmc.TicksPerSecondCommand
@sleep 3
fork instrumentation capabilities
@sleep 6
# The patch file is the same one server test 4 applies by redefinition.
fork instrumentation patch check tps-text retransform
@sleep 4
fork instrumentation patch apply tps-text retransform
@sleep 4
fork instrumentation patch apply tps-text retransform confirm
@sleep 4
tps
# Reading back is itself a retransformation: Lemon's transformer supplies the patch again.
fork instrumentation inspect org.spigotmc.TicksPerSecondCommand verify
@sleep 3
fork instrumentation patch verify tps-text
@sleep 3
tps
# Refused while the retransformation patch is active / never applied by retransformation.
fork instrumentation patch check tps-text redefine
@sleep 4
fork instrumentation patch check tps-structural retransform
@sleep 4
fork instrumentation history
@sleep 2
fork instrumentation patch rollback tps-text
@sleep 3
fork instrumentation patch rollback tps-text confirm
@sleep 3
tps
fork instrumentation status
