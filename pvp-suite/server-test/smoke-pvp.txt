# LemonPvP suite on a real server (docs/PVP-SUITE.md): module dependencies, services, kills
# (simulated with /pvp sim, the same code path as real kills), killstreak announcements,
# config validation incl. the potion effect registry, hot swap, statistics surviving a core
# reload, and a full GC after which no module class loader may be left.
fork module list

@sleep 2

@sleep 2

@sleep 2

pvp sim kill Alice Bob

pvp sim kill Bob Alice

@sleep 2

# config.yml checks: a rejected value, an unknown potion effect, then valid again
@shell sed -i 's/^k-factor: 32/k-factor: 500/' paperswap/module-data/pvp-core/config.yml
fork module config pvp-core
@sleep 2
@shell sed -i 's/^k-factor: 500/k-factor: 32/' paperswap/module-data/pvp-core/config.yml
@shell sed -i 's/"speed 1 10"/"flying 1 10"/' paperswap/module-data/pvp-killstreaks/config.yml
fork module config pvp-killstreaks
@sleep 2
@shell sed -i 's/"flying 1 10"/"speed 1 10"/' paperswap/module-data/pvp-killstreaks/config.yml
fork module config pvp-killstreaks
@sleep 2

# hot swap of a module that uses pvp-core, while it runs; the new instance announces
fork module reload pvp-killstreaks
@sleep 3
pvp sim kill Carol Dave
pvp sim kill Carol Dave
pvp sim kill Carol Dave
@sleep 1

# pvp-core cannot go away while modules use it
fork module disable pvp-core
@sleep 2

# statistics survive a reload of pvp-core (saved on disable, loaded on enable)
fork module unload pvp-scoreboard
@sleep 2
fork module unload pvp-killstreaks
@sleep 2
fork module unload pvp-combattag
@sleep 2
fork module reload pvp-core
@sleep 3
@shell grep -q "$(printf 'Alice\t5\t1\t0\t5\t1045')" paperswap/module-data/pvp-core/stats.tsv
stats Alice
top
@sleep 1

# everything off, full GC: every unloaded instance must be collected
fork module unload pvp-core
@sleep 3
say paperswap-gc-check
@shell jcmd "$SERVER_PID" GC.run
@sleep 2
fork live diagnostics
fork module diagnostics pvp-core
fork module diagnostics pvp-combattag
fork module diagnostics pvp-killstreaks
fork module diagnostics pvp-scoreboard
@sleep 3
@shell sed -n '/paperswap-gc-check/,$p' server.log > after-gc.log; grep -cE 'Unloaded [0-9.]+ at .*: class loader collected' after-gc.log; ! grep -nE 'still in memory|still reachable' after-gc.log
# pvp-core 2 (reload + unload), pvp-killstreaks 2 (reload + unload), pvp-combattag 1, pvp-scoreboard 1
@shell test "$(grep -cE 'Unloaded [0-9.]+ at .*: class loader collected' after-gc.log)" -eq 6
@shell if grep -qE 'still in memory|still reachable' after-gc.log; then jcmd "$SERVER_PID" GC.heap_dump "$PWD/leak.hprof" > /dev/null && java -Xmx3g "$SCRIPT_DIR/HeapPath.java" leak.hprof dev.paperswap.runtime.loader.ModuleClassLoader closed; rm -f leak.hprof; fi; true
